## 代码产物（验证正确）
`src/fib.py`:
```python
def fib(n: int) -> list[int]:
    """Return the first n terms of the Fibonacci sequence."""
    if n <= 0:
        return []
    elif n == 1:
        return [1]
    result = [1, 1]
    for i in range(2, n):
        result.append(result[-1] + result[-2])
    return result[:n]

def main():
    import sys
    if len(sys.argv) > 1:
        try:
            n = int(sys.argv[1])
            print(fib(n))
        except ValueError:
            print("Usage: python src.fib: fib <n>")
    else:
        print(fib(10))

if __name__ == "__main__":
    main()
```

`test_fib.py`:
```python
import pytest
from src.fib import fib

def test_fib_0(): assert fib(0) == []
def test_fib_1(): assert fib(1) == [1]
def test_fib_2(): assert fib(2) == [1, 1]
def test_fib_10():
    result = fib(10)
    assert len(result) == 10
    assert result[-1] == 55
```
